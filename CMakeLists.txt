cmake_minimum_required(VERSION 3.22)

# Must be set before project()
set(CMAKE_OSX_DEPLOYMENT_TARGET "10.15" CACHE STRING "Minimum macOS version")
set(CMAKE_OSX_ARCHITECTURES "arm64;x86_64" CACHE STRING "Universal binary (Apple Silicon + Intel)")

project(BeatCanvas VERSION 0.1.0 LANGUAGES C CXX)

set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)

include(FetchContent)
FetchContent_Declare(JUCE
    GIT_REPOSITORY https://github.com/juce-framework/JUCE.git
    GIT_TAG 8.0.4
    GIT_SHALLOW ON)
FetchContent_MakeAvailable(JUCE)

if(APPLE)
    set(BC_COPY TRUE)
    set(BC_FORMATS AU VST3 Standalone)      # AU is what Logic Pro loads
else()
    set(BC_COPY FALSE)
    set(BC_FORMATS VST3 Standalone)        # AU exists only on macOS; these let you iterate on Windows/Linux
endif()

juce_add_plugin(BeatCanvas
    COMPANY_NAME "Zevik"
    BUNDLE_ID com.zevik.beatcanvas
    PLUGIN_MANUFACTURER_CODE Zvkd           # 4 chars, at least one uppercase
    PLUGIN_CODE Bcnv                        # 4 chars, at least one uppercase
    FORMATS ${BC_FORMATS}
    PRODUCT_NAME "BeatCanvas"
    AU_MAIN_TYPE kAudioUnitType_Effect      # effect = loads on any track in Logic, passes audio through
    IS_SYNTH FALSE
    NEEDS_MIDI_INPUT FALSE
    NEEDS_MIDI_OUTPUT FALSE
    IS_MIDI_EFFECT FALSE
    COPY_PLUGIN_AFTER_BUILD ${BC_COPY})     # install into the user's plug-in folders (macOS only; Windows needs admin)

# ---- drop-in animations: every header in Source/Animations/ is picked up automatically ----
file(GLOB BC_ANIM_HEADERS CONFIGURE_DEPENDS ${CMAKE_CURRENT_SOURCE_DIR}/Source/Animations/*.h)
list(SORT BC_ANIM_HEADERS)
set(BC_INDEX "// Generated by CMake - do not edit.\n")
foreach(h ${BC_ANIM_HEADERS})
    get_filename_component(n ${h} NAME)
    string(APPEND BC_INDEX "#include \"Animations/${n}\"\n")
endforeach()
file(WRITE ${CMAKE_BINARY_DIR}/generated/AnimationIndex.inc.tmp "${BC_INDEX}")
configure_file(${CMAKE_BINARY_DIR}/generated/AnimationIndex.inc.tmp ${CMAKE_BINARY_DIR}/generated/AnimationIndex.inc COPYONLY)

target_sources(BeatCanvas PRIVATE
    Source/AnimationRegistry.cpp
    Source/PluginEditor.cpp
    Source/PluginProcessor.cpp
    Source/Presets.cpp)

target_include_directories(BeatCanvas PRIVATE Source ${CMAKE_BINARY_DIR}/generated)

target_compile_definitions(BeatCanvas PUBLIC
    JUCE_WEB_BROWSER=0
    JUCE_USE_CURL=0
    JUCE_VST3_CAN_REPLACE_VST2=0
    JUCE_DISPLAY_SPLASH_SCREEN=0)

target_link_libraries(BeatCanvas
    PRIVATE juce::juce_audio_utils
    PUBLIC  juce::juce_recommended_config_flags
            juce::juce_recommended_lto_flags
            juce::juce_recommended_warning_flags)

# ---- sync-engine unit test (pure C++, no JUCE) ----
enable_testing()
add_executable(ClockTest Tests/ClockTest.cpp)
target_include_directories(ClockTest PRIVATE Source)
add_test(NAME ClockTest COMMAND ClockTest)
